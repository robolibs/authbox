29pub fn contains_email_pattern(text: &str) -> bool {
30    text.split(|ch: char| {
31        ch.is_whitespace() || matches!(ch, '"' | '\'' | '<' | '>' | ',' | ';' | '#' | ':')
32    })
33    .any(token_looks_like_email)
34}
35
36fn token_looks_like_email(token: &str) -> bool {
37    let Some(at) = token.find('@') else {
38        return false;
39    };
40    if at == 0 || at + 1 >= token.len() || token[at + 1..].contains('@') {
41        return false;
42    }
43    let local = &token[..at];
44    let domain = &token[at + 1..];
45    if local.is_empty()
46        || !local
47            .chars()
48            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '%' | '+' | '-'))
49    {
50        return false;
51    }
52    let Some(dot) = domain.rfind('.') else {
53        return false;
54    };
55    dot > 0
56        && dot + 2 < domain.len() + 1
57        && domain[..dot]
58            .chars()
59            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '-'))
60        && domain[dot + 1..].chars().all(|c| c.is_ascii_alphabetic())
61        && domain[dot + 1..].len() >= 2
62}

64pub fn contains_phone_pattern(text: &str) -> bool {
65    let mut digits = 0usize;
66    let mut run_len = 0usize;
67    for ch in text.chars().chain(std::iter::once('x')) {
68        if ch.is_ascii_digit() || matches!(ch, '+' | '-' | '.' | ' ' | '\t' | '(' | ')') {
69            run_len += 1;
70            if ch.is_ascii_digit() {
71                digits += 1;
72            }
73            continue;
74        }
75        if digits >= 7 && run_len >= 7 {
76            return true;
77        }
78        digits = 0;
79        run_len = 0;
80    }
81    false
82}

84pub fn contains_name_pattern(text: &str) -> bool {
85    let words: Vec<&str> = text
86        .split(|ch: char| !ch.is_ascii_alphabetic())
87        .filter(|word| !word.is_empty())
88        .collect();
89    words
90        .windows(2)
91        .any(|pair| capitalized_word(pair[0]) && capitalized_word(pair[1]))
92}

94fn capitalized_word(word: &str) -> bool {
95    let trimmed = word.trim_matches(|ch: char| !ch.is_ascii_alphabetic());
96    let mut chars = trimmed.chars();
97    let Some(first) = chars.next() else {
98        return false;
99    };
100    first.is_ascii_uppercase()
101        && chars.clone().next().is_some()
102        && chars.all(|ch| ch.is_ascii_lowercase())
103}

105pub fn contains_address_pattern(text: &str) -> bool {
106    const SUFFIXES: &[&str] = &[
107        "Street",
108        "St",
109        "Avenue",
110        "Ave",
111        "Road",
112        "Rd",
113        "Drive",
114        "Dr",
115        "Boulevard",
116        "Blvd",
117    ];
118    let words: Vec<&str> = text.split_whitespace().collect();
119    words.windows(3).any(|triple| {
120        triple[0]
121            .trim_matches(|ch: char| !ch.is_ascii_alphanumeric())
122            .chars()
123            .all(|ch| ch.is_ascii_digit())
124            && capitalized_word(triple[1])
125            && SUFFIXES.contains(&triple[2].trim_matches(|ch: char| !ch.is_ascii_alphabetic()))
126    })
127}

129pub fn scan_did_for_pii(did: &str) -> Vec<PrivacyConcern> {
130    let mut concerns = Vec::new();
131    if contains_email_pattern(did) || did.contains("%40") {
132        concerns.push(PrivacyConcern {
133            risk: PrivacyRisk::Critical,
134            category: "Email Address".to_string(),
135            description:
136                "DID appears to contain an email address. This is personally identifiable information."
137                    .to_string(),
138            location: "id".to_string(),
139            recommendation: "Use a non-PII pairwise or pseudonymous identifier".to_string(),
140        });
141    }
142    if contains_phone_pattern(did) {
143        concerns.push(PrivacyConcern {
144            risk: PrivacyRisk::Critical,
145            category: "Phone Number".to_string(),
146            description:
147                "DID appears to contain a phone number. This is personally identifiable information."
148                    .to_string(),
149            location: "id".to_string(),
150            recommendation: "Do not put telephone numbers in immutable identifiers".to_string(),
151        });
152    }
153    if contains_name_pattern(did) {
154        concerns.push(PrivacyConcern {
155            risk: PrivacyRisk::High,
156            category: "Personal Name".to_string(),
157            description:
158                "DID appears to contain a personal name. This may be personally identifiable."
159                    .to_string(),
160            location: "id".to_string(),
161            recommendation: "Use a non-PII pairwise or pseudonymous identifier".to_string(),
162        });
163    }
164    concerns
165}

167pub fn scan_for_pii(doc: &DidDocument) -> Vec<PrivacyConcern> {
168    let mut concerns = scan_did_for_pii(&doc.id);
169    for (index, method) in doc.verification_methods.iter().enumerate() {
170        if contains_email_pattern(&method.id)
171            || contains_phone_pattern(&method.id)
172            || contains_name_pattern(&method.id)
173        {
174            concerns.push(PrivacyConcern {
175                risk: PrivacyRisk::High,
176                category: "PII in Verification Method".to_string(),
177                description:
178                    "Verification method ID may contain personally identifiable information."
179                        .to_string(),
180                location: format!("verificationMethod[{index}].id"),
181                recommendation: "Avoid embedding PII in verification method identifiers"
182                    .to_string(),
183            });
184        }
185    }
186    for (index, service) in doc.services.iter().enumerate() {
187        if contains_email_pattern(&service.id)
188            || contains_email_pattern(&service.service_endpoint_json)
189        {
190            concerns.push(PrivacyConcern {
191                risk: PrivacyRisk::Critical,
192                category: "Email in Service".to_string(),
193                description:
194                    "Service contains an email address. Avoid storing PII on immutable registries."
195                        .to_string(),
196                location: format!("service[{index}]"),
197                recommendation: "Remove personal email addresses from service entries".to_string(),
198            });
199        }
200        if contains_address_pattern(&service.service_endpoint_json) {
201            concerns.push(PrivacyConcern {
202                risk: PrivacyRisk::Critical,
203                category: "Physical Address".to_string(),
204                description: "Service endpoint contains what appears to be a physical address."
205                    .to_string(),
206                location: format!("service[{index}].serviceEndpoint"),
207                recommendation: "Remove physical addresses from service endpoints".to_string(),
208            });
209        }
210    }
211    concerns
212}

220pub fn get_max_risk(concerns: &[PrivacyConcern]) -> PrivacyRisk {
221    concerns
222        .iter()
223        .map(|concern| concern.risk.clone())
224        .max()
225        .unwrap_or(PrivacyRisk::None)

232pub fn check_correlation_risks(doc: &DidDocument) -> Vec<PrivacyConcern> {
233    let mut all_vm_refs = Vec::new();
234    all_vm_refs.extend(doc.authentication.iter().cloned());
235    all_vm_refs.extend(doc.assertion_method.iter().cloned());
236    all_vm_refs.extend(doc.key_agreement.iter().cloned());
237    all_vm_refs.extend(doc.capability_invocation.iter().cloned());
238    all_vm_refs.extend(doc.capability_delegation.iter().cloned());
239    let total = all_vm_refs.len();
240    all_vm_refs.sort();
241    all_vm_refs.dedup();
242    if all_vm_refs.len() < total {
243        vec![PrivacyConcern {
244            risk: PrivacyRisk::Medium,
245            category: "Key Reuse".to_string(),
246            description:
247                "Same verification method used for multiple purposes. Consider using separate keys for each purpose."
248                    .to_string(),
249            location: "verification relationships".to_string(),
250            recommendation: "Use separate keys for each verification relationship".to_string(),
251        }]
252    } else {
253        Vec::new()
254    }
255}

257pub fn detect_key_reuse(doc: &DidDocument) -> Vec<PrivacyConcern> {
258    check_correlation_risks(doc)
259}

261#[allow(dead_code)]
262fn reused_across_relationships(method: &VerificationMethod, doc: &DidDocument) -> bool {
263    let refs = [
264        &doc.authentication,
265        &doc.assertion_method,
266        &doc.key_agreement,
267        &doc.capability_invocation,
268        &doc.capability_delegation,
269    ];
270    refs.iter()
271        .filter(|items| items.iter().any(|item| item == &method.id))
272        .count()
273        > 1
274}

276pub fn format_concerns(concerns: &[PrivacyConcern]) -> String {
277    if concerns.is_empty() {
278        return "No privacy concerns detected.".to_string();
279    }
280    let mut result = "Privacy Concerns Detected:\n".to_string();
281    for concern in concerns {
282        result.push_str("\n[");
283        result.push_str(match concern.risk {
284            PrivacyRisk::None => "NONE",
285            PrivacyRisk::Low => "LOW",
286            PrivacyRisk::Medium => "MEDIUM",
287            PrivacyRisk::High => "HIGH",
288            PrivacyRisk::Critical => "CRITICAL",
289        });
290        result.push_str("] ");
291        result.push_str(&concern.category);
292        result.push_str("\n  Location: ");
293        result.push_str(&concern.location);
294        result.push_str("\n  Details: ");
295        result.push_str(&concern.description);
296        result.push('\n');
297    }
298    result
