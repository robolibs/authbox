3#[derive(Clone, Debug, Eq, PartialEq)]
4pub enum CryptographicStrength {
5    Deprecated,
6    Acceptable,
7    Recommended,
8    Experimental,
9}
10
11impl CryptographicStrength {
12    pub const DEPRECATED: Self = Self::Deprecated;
13    pub const ACCEPTABLE: Self = Self::Acceptable;
14    pub const RECOMMENDED: Self = Self::Recommended;
15    pub const EXPERIMENTAL: Self = Self::Experimental;
16}

18#[derive(Clone, Debug, Eq, PartialEq)]
19pub struct SecurityConcern {
20    pub strength: CryptographicStrength,
21    pub category: String,
22    pub description: String,
23    pub recommendation: String,
24    pub location: String,
25    pub method_id: String,
26}

28pub fn evaluate_key_type(type_: &str) -> CryptographicStrength {
29    if type_.contains("Ed25519") || type_.contains("EdDSA") {
30        return CryptographicStrength::Recommended;
31    }
32    if type_.contains("secp256k1") || type_.contains("EcdsaSecp256k1") {
33        return CryptographicStrength::Deprecated;
34    }
35    if type_.contains("P-256") || type_.contains("secp256r1") || type_.contains("ES256") {
36        return CryptographicStrength::Acceptable;
37    }
38    if type_.contains("P-384") || type_.contains("secp384r1") || type_.contains("ES384") {
39        return CryptographicStrength::Acceptable;
40    }
41    if type_.contains("P-521") || type_.contains("secp521r1") || type_.contains("ES512") {
42        return CryptographicStrength::Acceptable;
43    }
44    if type_.contains("RSA")
45        || type_.contains("RS256")
46        || type_.contains("RS384")
47        || type_.contains("RS512")
48    {
49        return CryptographicStrength::Acceptable;
50    }
51    if type_.contains("X25519") {
52        return CryptographicStrength::Recommended;
53    }
54    if type_.contains("Bls12381") {
55        return CryptographicStrength::Recommended;
56    }
57    if type_.contains("Dilithium") || type_.contains("Kyber") || type_.contains("Falcon") {
58        return CryptographicStrength::Experimental;
59    }
60    CryptographicStrength::Acceptable
61}

63pub fn evaluate_jwk_curve(crv: &str) -> CryptographicStrength {
64    match crv {
65        "Ed25519" | "Ed448" => CryptographicStrength::Recommended,
66        "P-256" | "secp256r1" => CryptographicStrength::Acceptable,
67        "P-384" | "secp384r1" => CryptographicStrength::Acceptable,
68        "P-521" | "secp521r1" => CryptographicStrength::Acceptable,
69        "secp256k1" => CryptographicStrength::Deprecated,
70        "X25519" | "X448" => CryptographicStrength::Recommended,
71        _ => CryptographicStrength::Acceptable,
72    }
73}

75pub fn evaluate_verification_method(method: &VerificationMethod, index: usize) -> SecurityConcern {
76    let mut strength = evaluate_key_type(&method.type_);
77    if let Some(jwk) = &method.public_key_jwk
78        && !jwk.crv.is_empty()
79    {
80        strength = evaluate_jwk_curve(&jwk.crv);
81    }
82    let (category, description, recommendation) = match strength {
83        CryptographicStrength::Deprecated => (
84            "Deprecated Cryptography",
85            format!("Uses deprecated cryptographic algorithm: {}", method.type_),
86            "Migrate to Ed25519 or other recommended algorithms. W3C DID Implementation Guide advises against secp256k1 and RSA < 2048.",
87        ),
88        CryptographicStrength::Acceptable => (
89            "Acceptable Cryptography",
90            format!(
91                "Uses acceptable but not recommended algorithm: {}",
92                method.type_
93            ),
94            "Consider Ed25519 for signatures or X25519 for key agreement in new implementations.",
95        ),
96        CryptographicStrength::Recommended => (
97            "Recommended Cryptography",
98            format!("Uses recommended cryptographic algorithm: {}", method.type_),
99            "No action needed. This is a recommended algorithm.",
100        ),
101        CryptographicStrength::Experimental => (
102            "Experimental Cryptography",
103            format!(
104                "Uses experimental cryptographic algorithm: {}",
105                method.type_
106            ),
107            "Ensure thorough testing and monitor for standardization progress.",
108        ),
109    };
110    SecurityConcern {
111        strength,
112        category: category.to_string(),
113        description,
114        recommendation: recommendation.to_string(),
115        location: format!("verificationMethod[{index}]"),
116        method_id: method.id.clone(),
117    }
118}

120pub fn scan_document(doc: &DidDocument) -> Vec<SecurityConcern> {
121    doc.verification_methods
122        .iter()
123        .enumerate()
124        .map(|(idx, method)| evaluate_verification_method(method, idx))
125        .filter(|concern| concern.strength != CryptographicStrength::Recommended)
126        .collect()
127}
128
129pub fn uses_recommended_cryptography(doc: &DidDocument) -> bool {
130    doc.verification_methods
131        .iter()
132        .all(|method| evaluate_key_type(&method.type_) != CryptographicStrength::Deprecated)
133}

135pub fn get_cryptography_summary(doc: &DidDocument) -> String {
136    let mut deprecated = 0;
137    let mut acceptable = 0;
138    let mut recommended = 0;
139    let mut experimental = 0;
140    for method in &doc.verification_methods {
141        match evaluate_key_type(&method.type_) {
142            CryptographicStrength::Deprecated => deprecated += 1,
143            CryptographicStrength::Acceptable => acceptable += 1,
144            CryptographicStrength::Recommended => recommended += 1,
145            CryptographicStrength::Experimental => experimental += 1,
146        }
147    }
148    format!(
149        "Cryptography Summary:\n  Recommended: {recommended}\n  Acceptable: {acceptable}\n  Deprecated: {deprecated}\n  Experimental: {experimental}\n"
150    )
151}

153pub fn format_concerns(concerns: &[SecurityConcern]) -> String {
154    if concerns.is_empty() {
155        return "No security concerns detected. All cryptography uses recommended algorithms."
156            .to_string();
157    }
158
159    let mut result = String::from("Security Concerns:\n");
160    for concern in concerns {
161        result.push_str("\n[");
162        result.push_str(match concern.strength {
163            CryptographicStrength::Deprecated => "DEPRECATED",
164            CryptographicStrength::Acceptable => "ACCEPTABLE",
165            CryptographicStrength::Recommended => "RECOMMENDED",
166            CryptographicStrength::Experimental => "EXPERIMENTAL",
167        });
168        result.push_str("] ");
169        result.push_str(&concern.category);
170        result.push('\n');
171        result.push_str("  Location: ");
172        result.push_str(&concern.location);
173        result.push('\n');
174        result.push_str("  Details: ");
175        result.push_str(&concern.description);
176        result.push('\n');
177        result.push_str("  Recommendation: ");
178        result.push_str(&concern.recommendation);
179        result.push('\n');
180    }
181
182    result
