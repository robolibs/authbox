19pub fn parse_did_peer(did_uri: &str) -> DidResult<Vec<PeerElement>> {
20    let parsed = parse(did_uri)?;
21    if parsed.method != "peer" {
22        return Err(error::invalid_method_id("method must be 'peer'"));
23    }
24    if !parsed.method_id.starts_with("2.") {
25        return Err(error::not_implemented("only did:peer:2 is supported"));
26    }
27    let mut out = Vec::new();
28    for element in parsed.method_id[2..].split('.') {
29        if element.is_empty() {
30            continue;
31        }
32        let (type_char, encoded_value) = element.split_at(1);
33        let type_ = match type_char {
34            "V" => PeerElementType::Verification,
35            "E" => PeerElementType::Encryption,
36            "S" => PeerElementType::Service,
37            _ => PeerElementType::Unknown,
38        };
39        out.push(PeerElement {
40            type_,
41            encoded_value: encoded_value.to_string(),
42        });
43    }
44    if out.is_empty() {
45        return Err(error::invalid_method_id(
46            "did:peer:2 must have at least one element",
47        ));
48    }
49    Ok(out)
50}

52pub fn decode_peer_key(encoded: &str) -> DidResult<Vec<u8>> {
53    if encoded.is_empty() {
54        return Err(error::invalid_key_format("empty encoded key"));
55    }
56    if !encoded.starts_with('z') {
57        return Err(error::invalid_key_format(
58            "only base58btc (z) encoding is supported",
59        ));
60    }
61    base58btc_decode(&encoded[1..])
62}

64pub fn resolve_did_peer_document_json(did_uri: &str) -> DidResult<String> {
65    let elements = parse_did_peer(did_uri)?;
66    let mut vm_entries = Vec::new();
67    let mut auth_refs = Vec::new();
68    let mut ka_refs = Vec::new();
69    let mut service_entries = Vec::new();
70    let mut vm_index = 0usize;
71    let mut service_index = 0usize;
72    for element in &elements {
73        match element.type_ {
74            PeerElementType::Verification | PeerElementType::Encryption => {
75                let key_bytes = match decode_peer_key(&element.encoded_value) {
76                    Ok(bytes) => bytes,
77                    Err(_) => continue,
78                };
79                if key_bytes.len() < 34 {
80                    continue;
81                }
82                let crv = match key_bytes[..2] {
83                    [0xed, 0x01] => "Ed25519",
84                    [0xec, 0x01] => "X25519",
85                    _ => continue,
86                };
87                let raw_key = &key_bytes[2..];
88                if raw_key.len() != 32 {
89                    continue;
90                }
91                let vm_id = format!("{did_uri}#key-{vm_index}");
92                vm_entries.push(format!(
93                    "    {{\n      \"id\": \"{vm_id}\",\n      \"type\": \"JsonWebKey2020\",\n      \"controller\": \"{did_uri}\",\n      \"publicKeyJwk\": {{\n        \"kty\": \"OKP\",\n        \"crv\": \"{crv}\",\n        \"x\": \"{}\"\n      }}\n    }}",
94                    base64url_encode(raw_key)
95                ));
96                vm_index += 1;
97            }
98            PeerElementType::Service => {
99                if let Some(entry) = service_entry(did_uri, &element.encoded_value, service_index) {
100                    service_entries.push(entry);
101                    service_index += 1;
102                }
103            }
104            _ => {}
105        }
106    }
107    // C++ compatibility: relationship references are assigned in a second pass
108    // from all V/E elements, even if the corresponding verificationMethod entry
109    // was skipped because its key could not be decoded.
110    let mut relationship_index = 0usize;
111    for element in &elements {
112        match element.type_ {
113            PeerElementType::Verification => {
114                auth_refs.push(format!("{did_uri}#key-{relationship_index}"));
115                relationship_index += 1;
116            }
117            PeerElementType::Encryption => {
118                ka_refs.push(format!("{did_uri}#key-{relationship_index}"));
119                relationship_index += 1;
120            }
121            _ => {}
122        }
123    }
124    let mut doc = format!(
125        "{{\n  \"@context\": [\n    \"https://www.w3.org/ns/did/v1\",\n    \"https://w3id.org/security/suites/jws-2020/v1\"\n  ],\n  \"id\": \"{did_uri}\",\n  \"verificationMethod\": [\n{}\n  ]",
126        vm_entries.join(",\n")
127    );
128    if !auth_refs.is_empty() {
129        doc.push_str(",\n  \"authentication\": [\n");
130        doc.push_str(
131            &auth_refs
132                .iter()
133                .map(|r| format!("    \"{r}\""))
134                .collect::<Vec<_>>()
135                .join(",\n"),
136        );
137        doc.push_str("\n  ]");
138    }
139    if !ka_refs.is_empty() {
140        doc.push_str(",\n  \"keyAgreement\": [\n");
141        doc.push_str(
142            &ka_refs
143                .iter()
144                .map(|r| format!("    \"{r}\""))
145                .collect::<Vec<_>>()
146                .join(",\n"),
147        );
148        doc.push_str("\n  ]");
149    }
150    if !service_entries.is_empty() {
151        doc.push_str(",\n  \"service\": [\n");
152        doc.push_str(&service_entries.join(",\n"));
153        doc.push_str("\n  ]");
154    }
155    doc.push_str("\n}\n");
156    Ok(doc)
157}

159fn service_entry(did_uri: &str, encoded_value: &str, service_index: usize) -> Option<String> {
160    let decoded = base64url_decode(encoded_value).ok()?;
161    let service_json = std::str::from_utf8(&decoded).ok()?;
162    let Json::Object(object) = parse_json(service_json).ok()? else {
163        return None;
164    };
165    let id = json_string_field(&object, "id")
166        .or_else(|| json_string_field(&object, "i"))
167        .unwrap_or_else(|| format!("#service-{service_index}"));
168    let type_ = json_string_field(&object, "type")
169        .or_else(|| json_string_field(&object, "t"))
170        .unwrap_or_else(|| "DIDCommMessaging".to_string());
171    let endpoint = object
172        .get("serviceEndpoint")
173        .or_else(|| object.get("s"))
174        .map(to_compact_string)?;
175    let id = if id.starts_with('#') {
176        format!("{did_uri}{id}")
177    } else {
178        id
179    };
180    Some(format!(
181        "    {{\n      \"id\": {},\n      \"type\": {},\n      \"serviceEndpoint\": {}\n    }}",
182        to_compact_string(&Json::String(id)),
183        to_compact_string(&Json::String(type_)),
184        endpoint
185    ))
186}

188fn json_string_field(
189    object: &std::collections::BTreeMap<String, Json>,
190    key: &str,
191) -> Option<String> {
192    match object.get(key) {
193        Some(Json::String(text)) => Some(text.clone()),
194        _ => None,
195    }
