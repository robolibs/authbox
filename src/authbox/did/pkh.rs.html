11pub fn parse_did_pkh(did_uri: &str) -> DidResult<PkhComponents> {
12    let parsed = parse(did_uri)?;
13    if parsed.method != "pkh" {
14        return Err(error::invalid_method_id("method must be 'pkh'"));
15    }
16    if parsed.method_id.is_empty() {
17        return Err(error::invalid_method_id("did:pkh method-id is empty"));
18    }
19    let parts: Vec<&str> = parsed.method_id.splitn(3, ':').collect();
20    if parts.len() != 3 {
21        return Err(error::invalid_method_id(
22            "did:pkh method-id must contain at least two colons",
23        ));
24    }
25    let namespace_id = parts[0];
26    let chain_id = parts[1];
27    let address = parts[2];
28    if namespace_id.len() < 3 || namespace_id.len() > 8 {
29        return Err(error::invalid_method_id("namespace must be 3-8 characters"));
30    }
31    if !namespace_id
32        .chars()
33        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit())
34    {
35        return Err(error::invalid_method_id(
36            "namespace must be lowercase alphanumeric",
37        ));
38    }
39    if chain_id.is_empty() || chain_id.len() > 32 {
40        return Err(error::invalid_method_id("chain_id must be 1-32 characters"));
41    }
42    if !chain_id
43        .chars()
44        .all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '-'))
45    {
46        return Err(error::invalid_method_id(
47            "chain_id must be alphanumeric with . or -",
48        ));
49    }
50    if address.is_empty() || address.len() > 64 {
51        return Err(error::invalid_method_id("address must be 1-64 characters"));
52    }
53    if !address
54        .chars()
55        .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.'))
56    {
57        return Err(error::invalid_method_id(
58            "address contains invalid characters",
59        ));
60    }
61    Ok(PkhComponents {
62        namespace_id: namespace_id.to_string(),
63        chain_id: chain_id.to_string(),
64        address: address.to_string(),
65    })
66}

75pub fn get_chain_name(components: &PkhComponents) -> String {
76    match (
77        components.namespace_id.as_str(),
78        components.chain_id.as_str(),
79    ) {
80        ("eip155", "1") => "Ethereum Mainnet".to_string(),
81        ("eip155", "5") => "Ethereum Goerli".to_string(),
82        ("eip155", "137") => "Polygon Mainnet".to_string(),
83        ("eip155", "8453") => "Base".to_string(),
84        ("eip155", chain) => format!("Ethereum Chain {chain}"),
85        ("bip122", _) => "Bitcoin".to_string(),
86        ("cosmos", chain) => format!("Cosmos {chain}"),
87        _ => format!("{}:{}", components.namespace_id, components.chain_id),
88    }
89}

99pub fn verify_ethereum_signature(address: &str, message: &str, signature: &str) -> DidResult<bool> {
100    let expected_address = normalize_ethereum_address(address)?;
101    let signature = decode_prefixed_hex(signature)?;
102    if signature.len() != secp256k1::RECOVERABLE_SIGNATURE_BYTES {
103        return Err(error::invalid_key_length(
104            secp256k1::RECOVERABLE_SIGNATURE_BYTES,
105            signature.len(),
106        ));
107    }
108
109    let normalized_v = secp256k1::normalize_recovery_id(signature[64]);
110    if !normalized_v.success {
111        return Err(error::invalid_key_format(normalized_v.error_message));
112    }
113
114    let digest = ethereum_personal_message_digest(message.as_bytes());
115    let recovered =
116        secp256k1::recover_public_key(&digest, &signature[..64], normalized_v.recovery_id);
117    if !recovered.success {
118        return Err(error::invalid_key_format(recovered.error_message));
119    }
120
121    let recovered_address = Context::to_hex(&secp256k1::ethereum_address_from_uncompressed_pubkey(
122        &recovered.public_key_uncompressed,
123    ));
124    Ok(recovered_address == expected_address)
125}

127fn normalize_ethereum_address(address: &str) -> DidResult<String> {
128    let stripped = address
129        .strip_prefix("0x")
130        .or_else(|| address.strip_prefix("0X"))
131        .ok_or_else(|| error::invalid_key_format("Ethereum address must start with 0x"))?;
132    if stripped.len() != 40 {
133        return Err(error::invalid_key_format(
134            "Ethereum address must contain 20 bytes",
135        ));
136    }
137    if !stripped.bytes().all(|byte| byte.is_ascii_hexdigit()) {
138        return Err(error::invalid_key_format(
139            "Ethereum address contains non-hex characters",
140        ));
141    }
142    Ok(stripped.to_ascii_lowercase())
143}

145fn decode_prefixed_hex(value: &str) -> DidResult<Vec<u8>> {
146    let stripped = value
147        .strip_prefix("0x")
148        .or_else(|| value.strip_prefix("0X"))
149        .unwrap_or(value);
150    if !stripped.len().is_multiple_of(2) {
151        return Err(error::invalid_key_format("hex value has odd length"));
152    }
153    let mut out = Vec::with_capacity(stripped.len() / 2);
154    for pair in stripped.as_bytes().chunks_exact(2) {
155        let high = hex_value(pair[0])
156            .ok_or_else(|| error::invalid_key_format("hex value contains non-hex characters"))?;
157        let low = hex_value(pair[1])
158            .ok_or_else(|| error::invalid_key_format("hex value contains non-hex characters"))?;
159        out.push((high << 4) | low);
160    }
161    Ok(out)
162}

164fn hex_value(byte: u8) -> Option<u8> {
165    match byte {
166        b'0'..=b'9' => Some(byte - b'0'),
167        b'a'..=b'f' => Some(10 + byte - b'a'),
168        b'A'..=b'F' => Some(10 + byte - b'A'),
169        _ => None,
170    }
171}

173fn ethereum_personal_message_digest(message: &[u8]) -> [u8; 32] {
174    let prefix = format!("\x19Ethereum Signed Message:\n{}", message.len());
175    let mut framed = Vec::with_capacity(prefix.len() + message.len());
176    framed.extend_from_slice(prefix.as_bytes());
177    framed.extend_from_slice(message);
178    keylock::keccak256(&framed)
179}
