44pub struct CrlDerCursor<'a> {
45    data: &'a [u8],
46    offset: usize,

49impl<'a> CrlDerCursor<'a> {
50    pub fn new(data: &'a [u8]) -> Self {
51        Self { data, offset: 0 }
52    }
53
54    pub fn remaining(&self) -> &'a [u8] {
55        &self.data[self.offset..]
56    }
57
58    pub fn empty(&self) -> bool {
59        self.offset >= self.data.len()
60    }
61
62    pub fn offset(&self) -> usize {
63        self.offset
64    }
65
66    pub fn advance(&mut self, bytes: usize) -> bool {
67        if self.offset + bytes > self.data.len() {
68            return false;
69        }
70        self.offset += bytes;
71        true
72    }
73}

165    pub fn verify_signature(&self, issuer: &Certificate) -> PkiResult<bool> {
166        if !matches!(
167            self.outer_signature.signature,
168            SignatureAlgorithmId::Ed25519
169                | SignatureAlgorithmId::RsaPkcs1Sha256
170                | SignatureAlgorithmId::RsaPkcs1Sha384
171                | SignatureAlgorithmId::RsaPkcs1Sha512
172                | SignatureAlgorithmId::RsaPssSha256
173                | SignatureAlgorithmId::RsaPssSha384
174                | SignatureAlgorithmId::RsaPssSha512
175                | SignatureAlgorithmId::EcdsaSha256
176                | SignatureAlgorithmId::EcdsaSha384
177                | SignatureAlgorithmId::EcdsaSha512
178        ) {
179            return Err(PkiError::new(
180                "Unsupported signature algorithm for CRL verification",
181            ));
182        }
183        verify_signature_bytes(
184            self.outer_signature.signature,
185            &self.tbs_der,
186            &self.signature_value,
187            &issuer.tbs.subject_public_key_info.public_key,
188        )
189    }

198    pub fn sign(&self, issuer_key: &KeyPair) -> PkiResult<Vec<u8>> {
199        match self.outer_signature.signature {
200            SignatureAlgorithmId::Ed25519 => {
201                sign_ed25519_detached(&self.tbs_der, &issuer_key.private_key)
202            }
203            SignatureAlgorithmId::Ed448 => {
204                sign_ed448_detached(&self.tbs_der, &issuer_key.private_key)
205            }
206            SignatureAlgorithmId::EcdsaSha256 => {
207                sign_ecdsa_p256_sha256(&self.tbs_der, &issuer_key.private_key)
208            }
209            SignatureAlgorithmId::EcdsaSha384 => {
210                sign_ecdsa_p384_sha384(&self.tbs_der, &issuer_key.private_key)
211            }
212            SignatureAlgorithmId::EcdsaSha512 => {
213                sign_ecdsa_p521_sha512(&self.tbs_der, &issuer_key.private_key)
214            }
215            SignatureAlgorithmId::RsaPkcs1Sha256
216            | SignatureAlgorithmId::RsaPkcs1Sha384
217            | SignatureAlgorithmId::RsaPkcs1Sha512 => {
218                sign_rsa_pkcs1v15_keypair(self.outer_signature.signature, &self.tbs_der, issuer_key)
219            }
220            SignatureAlgorithmId::RsaPssSha256
221            | SignatureAlgorithmId::RsaPssSha384
222            | SignatureAlgorithmId::RsaPssSha512 => {
223                sign_rsa_pss_keypair(self.outer_signature.signature, &self.tbs_der, issuer_key)
224            }
225            _ => Err(PkiError::new("Unsupported CRL signature algorithm")),
226        }
227    }

243fn compare_der_time(a: &DerTime, b: &DerTime) -> std::cmp::Ordering {
244    (a.year, a.month, a.day, a.hour, a.minute, a.second)
245        .cmp(&(b.year, b.month, b.day, b.hour, b.minute, b.second))
246}

248fn current_der_time_utc() -> DerTime {
249    let now = std::time::SystemTime::now()
250        .duration_since(std::time::UNIX_EPOCH)
251        .unwrap_or_default();
252    let seconds = now.as_secs() as i64;
253    let days = seconds / 86_400;
254    let seconds_of_day = seconds % 86_400;
255    let (year, month, day) = civil_from_days(days);
256    DerTime {
257        year,
258        month,
259        day,
260        hour: (seconds_of_day / 3_600) as u8,
261        minute: ((seconds_of_day % 3_600) / 60) as u8,
262        second: (seconds_of_day % 60) as u8,
263    }
264}

266fn civil_from_days(days_since_unix_epoch: i64) -> (i32, u8, u8) {
267    let z = days_since_unix_epoch + 719_468;
268    let era = if z >= 0 { z } else { z - 146_096 } / 146_097;
269    let doe = z - era * 146_097;
270    let yoe = (doe - doe / 1_460 + doe / 36_524 - doe / 146_096) / 365;
271    let y = yoe + era * 400;
272    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
273    let mp = (5 * doy + 2) / 153;
274    let day = doy - (153 * mp + 2) / 5 + 1;
275    let month = mp + if mp < 10 { 3 } else { -9 };
276    let year = y + i64::from(month <= 2);
277    (year as i32, month as u8, day as u8)
278}

307fn parse_crl_inner(input: &[u8], relaxed: bool) -> PkiResult<Crl> {
308    let top =
309        parse_sequence(input).map_err(|err| PkiError::new(format!("CRL outer sequence: {err}")))?;
310    let mut top_cursor = Cursor::new(&top.value);
311
312    let tbs = parse_sequence(top_cursor.remaining())?;
313    let tbs_der = top_cursor.remaining()[..tbs.bytes_consumed].to_vec();
314    top_cursor.advance(tbs.bytes_consumed)?;
315
316    let outer_signature = crl_parse_algorithm_identifier(top_cursor.remaining())?;
317    top_cursor.advance(outer_signature.bytes_consumed)?;
318
319    let signature = parse_bit_string(top_cursor.remaining())?;
320    top_cursor.advance(signature.bytes_consumed)?;
321    if !relaxed && !top_cursor.empty() {
322        return Err(PkiError::new("unexpected trailing data in CRL"));
323    }
324
325    let mut crl = Crl {
326        version: 1,
327        outer_signature: outer_signature.value,
328        signature_value: signature.value.bytes,
329        der: input[..top.bytes_consumed].to_vec(),
330        tbs_der,
331        ..Crl::default()
332    };
333
334    parse_tbs_crl(&tbs.value, &mut crl)?;
335    Ok(crl)
336}

338fn parse_tbs_crl(input: &[u8], crl: &mut Crl) -> PkiResult<()> {
339    let mut cursor = Cursor::new(input);
340    if !cursor.empty() {
341        let header = parse_id_len(cursor.remaining())?;
342        if header.value.identifier.tag_class == Asn1Class::Universal
343            && header.value.identifier.tag_number == Asn1Tag::Integer as u32
344        {
345            let version = parse_integer(cursor.remaining())?;
346            crl.version = integer_to_i32(&version.value)? + 1;
347            cursor.advance(version.bytes_consumed)?;
348        } else if header.value.identifier.tag_class == Asn1Class::ContextSpecific
349            && header.value.identifier.tag_number == 0
350        {
351            let start = header.value.header_bytes;
352            let end = start + header.value.length;
353            let version = parse_integer(&cursor.remaining()[start..end])?;
354            crl.version = integer_to_i32(&version.value)? + 1;
355            cursor.advance(header.bytes_consumed)?;
356        }
357    }
358
359    let signature = crl_parse_algorithm_identifier(cursor.remaining())?;
360    crl.signature = signature.value;
361    cursor.advance(signature.bytes_consumed)?;
362
363    let issuer = crl_parse_name(cursor.remaining())?;
364    crl.issuer = issuer.value;
365    cursor.advance(issuer.bytes_consumed)?;
366
367    let this_update = crl_parse_time(cursor.remaining())?;
368    crl.this_update = this_update.value;
369    cursor.advance(this_update.bytes_consumed)?;
370
371    if !cursor.empty() && is_time_tag(cursor.remaining()[0]) {
372        let next_update = crl_parse_time(cursor.remaining())?;
373        crl.next_update = Some(next_update.value);
374        cursor.advance(next_update.bytes_consumed)?;
375    }
376
377    if !cursor.empty() {
378        let header = parse_id_len(cursor.remaining())?;
379        if header.value.identifier.tag_class == Asn1Class::Universal
380            && header.value.identifier.tag_number == Asn1Tag::Sequence as u32
381        {
382            let revoked = parse_revoked_certificates(cursor.remaining())?;
383            crl.revoked = revoked.value;
384            cursor.advance(revoked.bytes_consumed)?;
385        }
386    }
387
388    if !cursor.empty()
389        && crl.version >= 2
390        && let Ok(extensions) = parse_crl_extensions(cursor.remaining(), crl)
391    {
392        crl.extensions = extensions.value;
393        cursor.advance(extensions.bytes_consumed)?;
394    }
395
396    Ok(())
397}

399fn integer_to_i32(bytes: &[u8]) -> PkiResult<i32> {
400    if bytes.is_empty() || bytes.len() > 4 {
401        return Err(PkiError::new("integer out of range"));
402    }
403    let mut out = 0i32;
404    for byte in bytes {
405        out = (out << 8) | i32::from(*byte);
406    }
407    Ok(out)
408}

423pub fn crl_parse_algorithm_identifier(input: &[u8]) -> PkiResult<Asn1Read<AlgorithmIdentifier>> {
424    let seq = parse_sequence(input)?;

433        identifier.hash = hash;

438pub fn crl_parse_time(input: &[u8]) -> PkiResult<Asn1Read<DerTime>> {
439    if input.is_empty() {
440        return Err(PkiError::new("empty time"));
441    }
442    match input[0] & 0x1f {
443        tag if tag == Asn1Tag::UtcTime as u8 => parse_utc_time(input),
444        tag if tag == Asn1Tag::GeneralizedTime as u8 => parse_generalized_time(input),
445        _ => Err(PkiError::new("unsupported time type")),
446    }
447}

449fn is_time_tag(byte: u8) -> bool {
450    let tag = byte & 0x1f;
451    tag == Asn1Tag::UtcTime as u8 || tag == Asn1Tag::GeneralizedTime as u8
452}

454pub fn parse_reason_code(input: &[u8]) -> Option<CrlReason> {
455    if input.len() != 3 || input[0] != Asn1Tag::Enumerated as u8 || input[1] != 0x01 {
456        return None;
457    }
458    match input[2] {
459        0 => Some(CrlReason::Unspecified),
460        1 => Some(CrlReason::KeyCompromise),
461        2 => Some(CrlReason::CaCompromise),
462        3 => Some(CrlReason::AffiliationChanged),
463        4 => Some(CrlReason::Superseded),
464        5 => Some(CrlReason::CessationOfOperation),
465        6 => Some(CrlReason::CertificateHold),
466        8 => Some(CrlReason::RemoveFromCrl),
467        9 => Some(CrlReason::PrivilegeWithdrawn),
468        10 => Some(CrlReason::AaCompromise),
469        _ => None,
470    }
471}

496pub fn parse_crl_entry_extensions(
497    input: &[u8],
498    revoked: &mut RevokedCertificate,
499) -> PkiResult<Asn1Read<Vec<CrlEntryExtension>>> {
500    let seq = parse_sequence(input)?;
501    let mut cursor = Cursor::new(&seq.value);
502    let mut extensions = Vec::new();
503    while !cursor.empty() {
504        let ext_seq = parse_sequence(cursor.remaining())?;
505        let mut ext_cursor = Cursor::new(&ext_seq.value);
506        let oid = parse_oid(ext_cursor.remaining())?;
507        ext_cursor.advance(oid.bytes_consumed)?;
508        let mut critical = false;
509        if !ext_cursor.empty() {
510            let next = parse_id_len(ext_cursor.remaining())?;
511            if next.value.identifier.tag_class == Asn1Class::Universal
512                && next.value.identifier.tag_number == Asn1Tag::Boolean as u32
513            {
514                let parsed = parse_boolean(ext_cursor.remaining())?;
515                critical = parsed.value;
516                ext_cursor.advance(parsed.bytes_consumed)?;
517            }
518        }
519        let value = parse_octet_string(ext_cursor.remaining())?;
520        ext_cursor.advance(value.bytes_consumed)?;
521        let id = identify_crl_entry_extension(&oid.value);
522        match id {
523            CrlEntryExtensionId::ReasonCode => revoked.reason = parse_reason_code(&value.value),
524            CrlEntryExtensionId::InvalidityDate => {
525                if let Ok(time) = parse_generalized_time(&value.value) {
526                    revoked.invalidity_date = Some(time.value);
527                }
528            }
529            CrlEntryExtensionId::CertificateIssuer => {
530                if let Ok(name) = crl_parse_name(&value.value) {
531                    revoked.certificate_issuer = Some(name.value);
532                }
533            }
534            CrlEntryExtensionId::Unknown => {}
535        }
536        extensions.push(CrlEntryExtension {
537            id,
538            oid: oid.value,
539            critical,
540            value: value.value,
541        });
542        cursor.advance(ext_seq.bytes_consumed)?;
543    }
544    Ok(Asn1Read::new(extensions, seq.bytes_consumed))
545}

547fn parse_revoked_certificates(input: &[u8]) -> PkiResult<Asn1Read<Vec<RevokedCertificate>>> {
548    let seq = parse_sequence(input)?;
549    let mut cursor = Cursor::new(&seq.value);
550    let mut revoked = Vec::new();
551    while !cursor.empty() {
552        let Ok(entry_seq) = parse_sequence(cursor.remaining()) else {
553            break;
554        };
555        let mut entry_cursor = Cursor::new(&entry_seq.value);
556        let serial = parse_integer(entry_cursor.remaining())
557            .map_err(|err| PkiError::new(format!("failed to parse revoked cert serial: {err}")))?;
558        entry_cursor.advance(serial.bytes_consumed)?;
559        let revocation_date = crl_parse_time(entry_cursor.remaining())
560            .map_err(|err| PkiError::new(format!("failed to parse revocation time: {err}")))?;
561        entry_cursor.advance(revocation_date.bytes_consumed)?;
562        let mut entry = RevokedCertificate {
563            serial_number: serial.value,
564            revocation_date: revocation_date.value,
565            ..RevokedCertificate::default()
566        };
567        if !entry_cursor.empty()
568            && let Ok(extensions) = parse_crl_entry_extensions(entry_cursor.remaining(), &mut entry)
569        {
570            entry.extensions = extensions.value;
571        }
572        revoked.push(entry);
573        cursor.advance(entry_seq.bytes_consumed)?;
574    }
575    Ok(Asn1Read::new(revoked, seq.bytes_consumed))
576}

578pub fn parse_crl_extensions(input: &[u8], crl: &mut Crl) -> PkiResult<Asn1Read<Vec<CrlExtension>>> {
579    let explicit = parse_id_len(input)?;
580    if explicit.value.identifier.tag_class != Asn1Class::ContextSpecific
581        || explicit.value.identifier.tag_number != 0
582        || !explicit.value.identifier.constructed
583    {
584        return Err(PkiError::new(
585            "expected context-specific [0] tag for CRL extensions",
586        ));
587    }
588    let start = explicit.value.header_bytes;
589    let end = start + explicit.value.length;
590    let seq = parse_sequence(&input[start..end])?;
591    let mut cursor = Cursor::new(&seq.value);
592    let mut extensions = Vec::new();
593    while !cursor.empty() {
594        let ext_seq = parse_sequence(cursor.remaining())?;
595        let mut ext_cursor = Cursor::new(&ext_seq.value);
596        let oid = parse_oid(ext_cursor.remaining())?;
597        ext_cursor.advance(oid.bytes_consumed)?;
598        let mut critical = false;
599        if !ext_cursor.empty() {
600            let next = parse_id_len(ext_cursor.remaining())?;
601            if next.value.identifier.tag_class == Asn1Class::Universal
602                && next.value.identifier.tag_number == Asn1Tag::Boolean as u32
603            {
604                let parsed = parse_boolean(ext_cursor.remaining())?;
605                critical = parsed.value;
606                ext_cursor.advance(parsed.bytes_consumed)?;
607            }
608        }
609        let value = parse_octet_string(ext_cursor.remaining())?;
610        ext_cursor.advance(value.bytes_consumed)?;
611        let id = identify_crl_extension(&oid.value);
612        match id {
613            CrlExtensionId::AuthorityKeyIdentifier => {
614                parse_authority_key_identifier(&value.value, crl)
615            }
616            CrlExtensionId::CrlNumber => {
617                if let Ok(parsed) = parse_integer(&value.value) {
618                    crl.crl_number = Some(parsed.value);
619                }
620            }
621            CrlExtensionId::DeltaCrlIndicator => {
622                if let Ok(parsed) = parse_integer(&value.value) {
623                    crl.delta_crl_indicator = Some(parsed.value);
624                }
625            }
626            CrlExtensionId::IssuingDistributionPoint => crl.has_issuing_distribution_point = true,
627            _ => {}
628        }
629        extensions.push(CrlExtension {
630            id,
631            oid: oid.value,
632            critical,
633            value: value.value,
634        });
635        cursor.advance(ext_seq.bytes_consumed)?;
636    }
637    Ok(Asn1Read::new(
638        extensions,
639        explicit.value.header_bytes + explicit.value.length,
640    ))
641}
642
643fn parse_authority_key_identifier(input: &[u8], crl: &mut Crl) {
644    let Ok(seq) = parse_sequence(input) else {
645        return;
646    };
647    let cursor = Cursor::new(&seq.value);
648    if cursor.empty() {
649        return;
650    }
651    let Ok(header) = parse_id_len(cursor.remaining()) else {
652        return;
653    };
654    if header.value.identifier.tag_class == Asn1Class::ContextSpecific
655        && header.value.identifier.tag_number == 0
656    {
657        let start = header.value.header_bytes;
658        let end = start + header.value.length;
659        crl.authority_key_identifier = Some(cursor.remaining()[start..end].to_vec());
660    }
661}

663pub fn parse_pem_crl_chain(pem: &str) -> PkiResult<Vec<Crl>> {
664    let mut out = Vec::new();
665    let mut pos = 0usize;
666    while let Some(rel_start) = pem[pos..].find("-----BEGIN") {
667        let start = pos + rel_start;
668        let tail = &pem[start..];
669        let block = super::pem_decode_block_with_expected_label(tail, Some("CRL"))
670            .or_else(|_| super::pem_decode_block_with_expected_label(tail, Some("X509 CRL")))?;
671        out.push(parse_crl(&block.data)?);
672        let Some(rel_end) = tail.find("-----END") else {
673            break;
674        };
675        pos = start + rel_end + 1;
676    }
677    if out.is_empty() {
678        return Err(PkiError::new("no CRLs found in PEM"));
679    }
680    Ok(out)
681}
682
683pub fn parse_pem_crl_chain_result(pem: &str) -> CertificateResult<Vec<Crl>> {
684    match parse_pem_crl_chain(pem) {
685        Ok(crls) => CertificateResult::ok(crls),
686        Err(error) => CertificateResult::failure(error.message),
687    }
688}
689
690pub fn load_crl(path: impl AsRef<Path>) -> PkiResult<Crl> {
691    let loaded = read_pem_or_der(path, None)?;
692    match loaded {
693        super::PemOrDer::Der(bytes) => parse_crl(&bytes),
694        super::PemOrDer::Pem(block) => {
695            if block.label == "CRL" || block.label == "X509 CRL" {
696                parse_crl(&block.data)
697            } else {
698                Err(PkiError::new("PEM block is not a CRL"))
699            }
700        }
701    }
702}
703
704pub fn load_crl_result(path: impl AsRef<Path>) -> CertificateResult<Crl> {
705    match load_crl(path) {
706        Ok(crl) => CertificateResult::ok(crl),
707        Err(error) => CertificateResult::failure(error.message),
708    }
709}

711pub fn encode_der_time(time: &DerTime) -> Vec<u8> {
712    if (1950..=2049).contains(&time.year) {
713        der::encode_utctime(&format!(
714            "{:02}{:02}{:02}{:02}{:02}{:02}Z",
715            time.year % 100,
716            time.month,
717            time.day,
718            time.hour,
719            time.minute,
720            time.second
721        ))
722    } else {
723        der::encode_generalized_time(&format!(
724            "{:04}{:02}{:02}{:02}{:02}{:02}Z",
725            time.year, time.month, time.day, time.hour, time.minute, time.second
726        ))
727    }
728}
